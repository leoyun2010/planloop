### 1.2 POL-019 의사코드 추가 (TO-BE)

```js
// 할인 승인 전 CCW (DEC-89) — Discount %·Extended Net Price 가 없고 Unit Net Price Before Credits 만 있을 때
function fromNetUnit(c, listAmount, currency) {
  const listUnit = num(c["Unit List Price"]), netUnit = num(c["Unit Net Price Before Credits"]);
  if (netUnit == null) return {discountRate: null, vendorNetAmount: null};
  if (!listUnit) return {discountRate: null, vendorNetAmount: roundNet(netUnit * qty, currency)};   // 번들 0원
  return {discountRate: round2((1 - netUnit / listUnit) * 100),
          vendorNetAmount: roundNet(listAmount * netUnit / listUnit, currency)};                   // KRW 원 단위
}
```
